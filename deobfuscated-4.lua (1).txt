local function v322()
	local t27 = {}

	if BackpackApp then
		if type(BackpackApp.button_by_unique) == "table" then
			for k, v in pairs(BackpackApp.button_by_unique) do
				local item_data = v.item_data or v.kind and v or nil

				if item_data then
					local category = item_data.category

					if category then
						local v328 = v51(item_data)
						local n9 = 0

						pcall(function()
							n9 = t1.val(v328, item_data.properties, category, item_data)
						end)

						if n9 > 0 then
							table.insert(t27, {
								unique = item_data.unique or k,
								kind = v328,
								value = n9,
								category = category
							})
						end
					end
				end
			end

			return t27
		end
	end

	return t27
end
local function v330()
	local t28 = {}
	local ok, result = pcall(function()
		return TradeApp:_get_my_offer()
	end)

	if ok and result then
		if type(result.items) == "table" then
			for _, item in ipairs(result.items) do
				if item.unique then
					t28[item.unique] = true
				end
			end
		end
	end

	return t28
end
local function v336(p66, p67, p68)
	local v337 = p68 or 0.14

	table.sort(p66, function(p69, p70)
		return p69.value > p70.value
	end)

	local t29 = {}
	local n10 = 0

	for _, v in ipairs(p66) do
		if #t29 >= 18 or p67 <= n10 then
			break
		end

		if n10 + v.value <= p67 * (1 + v337) then
			table.insert(t29, v)
			n10 += v.value
		end
	end

	return t29, n10
end

local u342 = false

_G.doAutoFill = function(p71)
	if not u342 then
		u342 = true
		task.spawn(function()
			pcall(function()
				TradeApp:refresh_all()
			end)
			task.wait(0.25)

			local _, v344 = t1.GetItemsFromPartnerOffer()
			local v345 = v344 or 0

			if v345 <= 0 then
				v321("Партнёр не добавил предметы!", Color3.fromRGB(200, 60, 60))
				u342 = false

				return
			end

			local v346 = t26[p71]
			local v347 = v345 * v346.ratio

			v321(string.format("Партнёр: %.2f → Цель: %.2f [%s]", v345, v347, v346.label), Color3.fromRGB(50, 50, 50))

			local ok, result = pcall(function()
				return TradeApp:_get_my_offer()
			end)

			if ok and result and type(result.items) == "table" and #result.items > 0 then
				local t30 = {}

				for _, item in ipairs(result.items) do
					table.insert(t30, item)
				end

				for _, v in ipairs(t30) do
					pcall(function()
						TradeApp:_remove_item_from_my_offer(v)
					end)
					task.wait(0.12)
				end

				task.wait(0.25)
			end

			local v355 = v322()
			local v356 = v330()
			local t31 = {}

			for _, v in ipairs(v355) do
				if not v356[v.unique] then
					table.insert(t31, v)
				end
			end

			if #t31 == 0 then
				v321("Нет предметов с известной ценностью!", Color3.fromRGB(200, 60, 60))
				u342 = false

				return
			end

			local v360, v361 = v336(t31, v347)

			if #v360 == 0 then
				v321(string.format("Нет подходящих под %.2f", v347), Color3.fromRGB(200, 60, 60))
				u342 = false

				return
			end

			local n11 = 0

			for _, v in ipairs(v360) do
				if pcall(function()
					RouterClient.get("TradeAPI/AddItemToOffer"):FireServer(v.unique)
				end) then
					n11 += 1
				end

				task.wait(0.12)
			end

			task.wait(0.3)
			pcall(function()
				TradeApp:refresh_all()
			end)
			task.wait(0.15)
			t18.Refresh()
			v321(string.format("✓ [%s] %d пред. | мой: %.2f / партнёр: %.2f", v346.label, n11, v361, v345), Color3.fromRGB(30, 140, 60))
			task.delay(5, function()
				v321("Нажми кнопку для авто-заполнения")
			end)
			u342 = false
		end)

		return
	end

	v321("Подождите, идёт заполнение...", Color3.fromRGB(200, 150, 30))
end
